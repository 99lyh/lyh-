// ========== 红外对管引脚定义 ==========
#define IR_LEFT_PIN   WIFI_IOT_IO_NAME_GPIO_13
#define IR_RIGHT_PIN  WIFI_IOT_IO_NAME_GPIO_14

// ========== 避险参数(可调) ==========
#define AVOID_REVERSE_US  800000    // 倒车时长(微秒), 800000=0.8秒
#define AVOID_TURN_US     1200000   // 转向调整车头时长(微秒), 1200000=1.2秒
#define AVOID_COOLDOWN_US 500000    // 避险后冷却(微秒), 防止刚恢复就重复触发

// ========== 全局变量 ==========
static void thread1(void);
static void thread2(void);

uint8_t uart_sendbuf[20];

// 紧急状态标志：1=正在紧急避让，thread1暂停
volatile int g_emergency_stop = 0;

// 互斥锁：保护串口发送，防止两个任务同时发送数据
osMutexId_t mutex_id;

/***通信协议***/
void stm32motor_control(int motorA, int motorB)
{
    uint8_t A_dir = 0;
    uint8_t B_dir = 0;

    if(motorA < 0){
        A_dir = 1;
        motorA = -motorA;
    }else{
        A_dir = 0;
    }
    
    if(motorB < 0){
        B_dir = 1;
        motorB = -motorB;
    }else{
        B_dir = 0;
    }
    
    if (motorA > 150) motorA = 150;
    if (motorB > 150) motorB = 150;

    uart_sendbuf[0] = 0xFC;
    uart_sendbuf[1] = A_dir;
    uart_sendbuf[2] = motorA;
    uart_sendbuf[3] = B_dir;
    uart_sendbuf[4] = motorB;
    uart_sendbuf[5] = 0xFD;
    
    // 加锁保护，防止两个任务同时发送串口数据
    osMutexAcquire(mutex_id, osWaitForever);
    UartWrite(WIFI_IOT_UART_IDX_2, (unsigned char *)uart_sendbuf, 6);
    osMutexRelease(mutex_id);
}

// ==================== 速度定义（全部减半） ====================
void car_backward(void) { stm32motor_control(-50, -50); }
void car_forward(void)  { stm32motor_control(50, 50); }
void car_left(void)     { stm32motor_control(25, 75); }
void car_right(void)    { stm32motor_control(75, 25); }
void car_stop(void)     { stm32motor_control(0, 0); }

// ========== 读取红外状态 ==========
int read_ir_left(void)
{
    WifiIotGpioValue val;
    GpioGetInputVal(IR_LEFT_PIN, &val);
    return (val == WIFI_IOT_GPIO_VALUE1) ? 1 : 0;
}

int read_ir_right(void)
{
    WifiIotGpioValue val;
    GpioGetInputVal(IR_RIGHT_PIN, &val);
    return (val == WIFI_IOT_GPIO_VALUE1) ? 1 : 0;
}

// ========== 红外对管初始化 ==========
void ir_sensor_init(void)
{
    IoSetFunc(IR_LEFT_PIN, WIFI_IOT_IO_FUNC_GPIO_13_GPIO);
    GpioSetDir(IR_LEFT_PIN, WIFI_IOT_GPIO_DIR_IN);
    
    IoSetFunc(IR_RIGHT_PIN, WIFI_IOT_IO_FUNC_GPIO_14_GPIO);
    GpioSetDir(IR_RIGHT_PIN, WIFI_IOT_GPIO_DIR_IN);
    
    printf("[IR] IR sensors initialized on GPIO13, GPIO14\r\n");
}

/***** 任务一：正常运动控制（优先级24） *****/
static void thread1(void)
{
    printf("[thread1] start motion sequence\r\n");
    while (1)
    {
        // ===== 等待紧急状态解除 =====
        if (g_emergency_stop)
        {
            usleep(50000);
            continue;
        }
        
        // ===== 前进 2秒 =====
        car_forward();
        printf("[thread1] FORWARD(50,50)\r\n");
        for (int i = 0; i < 20; i++) {
            if (g_emergency_stop) break;
            usleep(100000);
        }
        if (g_emergency_stop) continue;

        // ===== 左转 2秒 =====
        car_left();
        printf("[thread1] LEFT(25,75)\r\n");
        for (int i = 0; i < 20; i++) {
            if (g_emergency_stop) break;
            usleep(100000);
        }
        if (g_emergency_stop) continue;

        // ===== 右转 2秒 =====
        car_right();
        printf("[thread1] RIGHT(75,25)\r\n");
        for (int i = 0; i < 20; i++) {
            if (g_emergency_stop) break;
            usleep(100000);
        }
        if (g_emergency_stop) continue;

        // ===== 倒车 1秒 =====
        car_backward();
        printf("[thread1] BACKWARD(-50,-50)\r\n");
        for (int i = 0; i < 10; i++) {
            if (g_emergency_stop) break;
            usleep(100000);
        }
        if (g_emergency_stop) continue;

        // ===== 停止 1秒 =====
        car_stop();
        printf("[thread1] STOP(0,0)\r\n");
        for (int i = 0; i < 10; i++) {
            if (g_emergency_stop) break;
            usleep(100000);
        }
    }
}

/***** 任务二：边缘检测 + 紧急避让（优先级30，更高） *****/
static void thread2(void)
{
    printf("[thread2] start edge detection\r\n");
    while (1)
    {
        // ===== 读取红外状态 =====
        int left_edge = read_ir_left();
        int right_edge = read_ir_right();
        
        // 调试打印：每秒一次
        static int debug_counter = 0;
        debug_counter++;
        if (debug_counter >= 50) {
            debug_counter = 0;
            printf("[thread2] L=%d R=%d | emergency=%d\r\n", 
                   left_edge, right_edge, g_emergency_stop);
        }
        
        // ===== 检测到边缘 且 当前不处于紧急状态 =====
        if ((left_edge || right_edge) && (g_emergency_stop == 0))
        {
            printf("[thread2] !!! EDGE DETECTED !!! L=%d R=%d\r\n", left_edge, right_edge);
            
            // ---- 步骤1：设置紧急标志，暂停 thread1 ----
            g_emergency_stop = 1;

            // ---- 步骤2：急刹 ----
            car_stop();
            printf("[thread2] STOP(0,0)\r\n");
            usleep(100000);

            // ---- 步骤3：倒车（固定时长，远离边缘） ----
            car_backward();
            printf("[thread2] BACKWARD(-50,-50) [0.8s]\r\n");
            usleep(AVOID_REVERSE_US);

            // ---- 步骤4：转向调整车头方向（根据触发的传感器，朝远离边缘一侧转） ----
            if (left_edge && right_edge)
            {
                // 两侧都到边：向左转调整车头
                car_left();
                printf("[thread2] TURN LEFT(25,75) [1.2s]\r\n");
                usleep(AVOID_TURN_US);
            }
            else if (left_edge)
            {
                // 左侧到边：向右转（车头远离左侧边缘）
                car_right();
                printf("[thread2] TURN RIGHT(75,25) [1.2s]\r\n");
                usleep(AVOID_TURN_US);
            }
            else   // right_edge
            {
                // 右侧到边：向左转（车头远离右侧边缘）
                car_left();
                printf("[thread2] TURN LEFT(25,75) [1.2s]\r\n");
                usleep(AVOID_TURN_US);
            }

            // ---- 步骤5：停止 ----
            car_stop();
            printf("[thread2] STOP(0,0)\r\n");
            usleep(100000);

            // ---- 步骤6：解除紧急标志，恢复正常行驶 ----
            g_emergency_stop = 0;
            printf("[thread2] === AVOIDANCE DONE, resume normal ===\r\n");

            // ---- 步骤7：冷却一段时间，避免刚恢复就重复触发边缘检测 ----
            usleep(AVOID_COOLDOWN_US);
        }
        
        usleep(20000);  // 20ms 检测一次
    }
}

static void correspondence(void)
{
    printf("[correspondence] init start\r\n");
    GpioInit();
    
    // ========== UART2 初始化 ==========
    IoSetFunc(WIFI_IOT_IO_NAME_GPIO_11, WIFI_IOT_IO_FUNC_GPIO_11_UART2_TXD);
    IoSetFunc(WIFI_IOT_IO_NAME_GPIO_12, WIFI_IOT_IO_FUNC_GPIO_12_UART2_RXD);

    WifiIotUartAttribute uart_attr2 = {
        .baudRate = 115200,
        .dataBits = 8,
        .stopBits = 1,
        .parity = 0,
    };
    UartInit(WIFI_IOT_UART_IDX_2, &uart_attr2, NULL);
    printf("[correspondence] UART2 init ok (115200,8N1)\r\n");

    // ========== 红外初始化 ==========
    ir_sensor_init();
    printf("[correspondence] IR sensors initialized\r\n");

    // ========== 创建互斥锁 ==========
    mutex_id = osMutexNew(NULL);
    if (mutex_id == NULL)
    {
        printf("Failed to create Mutex!\n");
    }
    printf("[correspondence] mutex created\r\n");

    // ========== 创建任务 ==========
    osThreadAttr_t attr;
    attr.attr_bits = 0U;
    attr.cb_mem = NULL;
    attr.cb_size = 0U;
    attr.stack_mem = NULL;
    attr.stack_size = 1024 * 4;
    
    // 任务1：正常运动（优先级 24）
    attr.name = "thread1";
    attr.priority = 24;
    if (osThreadNew((osThreadFunc_t)thread1, NULL, &attr) == NULL)
    {
        printf("Failed to create thread1!\n");
    }
    printf("[correspondence] thread1 created (priority 24)\r\n");
    
    // 任务2：边缘检测（优先级 30，更高）
    attr.name = "thread2";
    attr.priority = 30;
    if (osThreadNew((osThreadFunc_t)thread2, NULL, &attr) == NULL)
    {
        printf("Failed to create thread2!\n");
    }
    printf("[correspondence] thread2 created (priority 30)\r\n");
    
    printf("[correspondence] init done\r\n");
}

APP_FEATURE_INIT(correspondence);
