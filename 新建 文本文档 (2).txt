#include <stdio.h>
#include <unistd.h>
#include "ohos_init.h"
#include "cmsis_os2.h"
#include "wifiiot_gpio.h"
#include "wifiiot_gpio_ex.h"
#include "hi_io.h"
#include "hi_time.h"
#include "hi_gpio.h"   // 补上这个

osMutexId_t mutex_id;
#define GPIO2 2
uint8_t flag = 0;

// 舵机控制函数（周期20ms）
void set_angle(unsigned int duty)
{
    // 确保引脚功能正确
    hi_io_set_func(HI_IO_NAME_GPIO_2, 0);  // GPIO模式
    GpioSetDir(GPIO2, WIFI_IOT_GPIO_DIR_OUT);

    GpioSetOutputVal(GPIO2, WIFI_IOT_GPIO_VALUE1);
    hi_udelay(duty);
    GpioSetOutputVal(GPIO2, WIFI_IOT_GPIO_VALUE0);
    hi_udelay(20000 - duty);
}

void engine_run_90(void)
{
    for (int i = 0; i < 20; i++) {   // 多发几次脉冲更稳
        set_angle(1500);
    }
}

void engine_run_180(void)
{
    for (int i = 0; i < 20; i++) {
        set_angle(2500);
    }
}

// 高优先级任务：转90度
void thread1(void *arg)
{
    (void)arg;
    while (1) {
        osMutexAcquire(mutex_id, osWaitForever);
        printf("thread1 is runing.\r\n");
        flag = 90;
        engine_run_90();
        osDelay(50);               // 持有锁一小段时间
        osMutexRelease(mutex_id);
        osDelay(100);
    }
}

// 中优先级任务：只监控打印（不抢锁）
void thread2(void *arg)
{
    (void)arg;
    osDelay(50);
    while (1) {
        printf("thread2 is runing.\r\n");
        if (flag == 90) {
            printf("SG90 turn 90 du.\r\n");
        } else if (flag == 180) {
            printf("SG90 turn 180 du.\r\n");
        }
        flag = 0;
        osDelay(100);
    }
}

// 低优先级任务：转180度
void thread3(void *arg)
{
    (void)arg;
    while (1) {
        osMutexAcquire(mutex_id, osWaitForever);
        printf("thread3 is runing.\r\n");
        flag = 180;
        engine_run_180();
        osDelay(200);              // 低优先级持有锁更久，体现互斥效果
        osMutexRelease(mutex_id);
        osDelay(100);
    }
}

static void SG90(void)
{
    // 创建互斥锁
    mutex_id = osMutexNew(NULL);
    if (mutex_id == NULL) {
        printf("Failed to create Mutex!\r\n");
        return;
    }

    osThreadAttr_t attr = {0};
    attr.stack_size = 1024 * 8;   // 加大堆栈，防止溢出

    // 任务1 高优先级
    attr.name = "thread1";
    attr.priority = osPriorityNormal + 2;   // 比普通高一点
    if (osThreadNew(thread1, NULL, &attr) == NULL) {
        printf("Failed to create thread1!\r\n");
    }

    // 任务2 中优先级
    attr.name = "thread2";
    attr.priority = osPriorityNormal;
    if (osThreadNew(thread2, NULL, &attr) == NULL) {
        printf("Failed to create thread2!\r\n");
    }

    // 任务3 低优先级
    attr.name = "thread3";
    attr.priority = osPriorityNormal - 2;
    if (osThreadNew(thread3, NULL, &attr) == NULL) {
        printf("Failed to create thread3!\r\n");
    }

    printf("SG90 Mutex tasks started!\r\n");
}

APP_FEATURE_INIT(SG90);
